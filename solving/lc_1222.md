# Queens That Can Attack the King

```java
class Solution {
    class Pair{
        private final int x;
        private final int y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }

        @Override
        public int hashCode() {
            return Objects.hash(x, y);
        }

        @Override
        public boolean equals(Object obj) {
            if (this == obj) return true;
            if (obj == null || getClass() != obj.getClass()) return false;
            Pair pair = (Pair) obj;
            return Objects.equals(x, pair.x) &&
                Objects.equals(y, pair.y);
        }
    }
    public List<List<Integer>> queensAttacktheKing(int[][] queens, int[] king) {
        HashSet<Pair> Queens = new HashSet<>();
        List<List<Integer>> ans = new ArrayList<>();
        for(int[] q : queens) Queens.add(new Pair(q[0], q[1]));
        // 1st direction TOP
        for(int y=king[1]; y>=0; y--){
            if(Queens.contains(new Pair(king[0], y))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(king[0]);
                queen.add(y);
                ans.add(queen);
                break;
            }
        }
        // 2nd direction TOP-RIGHT
        int x = king[0], y = king[1];
        while(x < 8 && y >= 0){
            if(Queens.contains(new Pair(x, y))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(x);
                queen.add(y);
                ans.add(queen);
                break;
            }
            x++; y--;
        }
        // 3rd direction RIGHT
        for(int X=king[0]; X<8; X++){
            if(Queens.contains(new Pair(X, king[1]))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(X);
                queen.add(king[1]);
                ans.add(queen);
                break;
            }
        }
        // 4rth direction BOTTOM-RIGHT
        x = king[0]; y = king[1];
        while(x < 8 && y < 8){
            if(Queens.contains(new Pair(x, y))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(x);
                queen.add(y);
                ans.add(queen);
                break;
            }
            x++; y++;
        }
        // 5th direction BOTTOM
        for(int Y=king[1]; Y<8; Y++){
            if(Queens.contains(new Pair(king[0], Y))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(king[0]);
                queen.add(Y);
                ans.add(queen);
                break;
            }
        }
        // 6th direction BOTTOM-LEFT
        x = king[0]; y = king[1];
        while(x >= 0 && y < 8){
            if(Queens.contains(new Pair(x, y))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(x);
                queen.add(y);
                ans.add(queen);
                break;
            }
            x--; y++;
        }
        // 7th direction LEFT
        for(int X=king[0]; X>=0; X--){
            if(Queens.contains(new Pair(X, king[1]))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(X);
                queen.add(king[1]);
                ans.add(queen);
                break;
            }
        }
        // 8th direction TOP-LEFT
        x = king[0]; y = king[1];
        while(x >= 0 && y >= 0){
            if(Queens.contains(new Pair(x, y))){
                ArrayList<Integer> queen = new ArrayList<>();
                queen.add(x);
                queen.add(y);
                ans.add(queen);
                break;
            }
            x--; y--;
        }
        return ans;
    }
}
```
