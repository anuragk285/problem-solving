# Maximum Non Negative Product in a Matrix

```java
class Solution {
    int[][] grid;
    long[][][] dp;
    int m, n;
    public int maxProductPath(int[][] grid) {
        this.grid = grid;
        m = grid.length;
        n = grid[0].length;
        dp = new long[m][n][2];
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                long[] min_max = getMinMax(i, j);
                dp[i][j][0] = min_max[0];
                dp[i][j][1] = min_max[1];
            }
        }
        if(dp[m-1][n-1][1] < 0) return -1;
        return (int)(dp[m-1][n-1][1] % 1_000_000_007);
    }
    private long[] getMinMax(int i, int j){
        int val = grid[i][j];
        long min = Integer.MAX_VALUE, max = Integer.MIN_VALUE;
        if(i == 0){
            if(j == 0) return new long[]{val, val};
            else{
                min = Math.min(val*dp[i][j-1][0], val*dp[i][j-1][1]);
                max = Math.max(val*dp[i][j-1][0], val*dp[i][j-1][1]);
            }
        }else{
            if(j == 0){
                min = Math.min(val*dp[i-1][j][0], val*dp[i-1][j][1]);
                max = Math.max(val*dp[i-1][j][0], val*dp[i-1][j][1]);
            }
            else{
                min = Math.min(val*dp[i-1][j][0], val*dp[i-1][j][1]);
                min = Math.min(min, val*dp[i][j-1][0]);
                min = Math.min(min, val*dp[i][j-1][1]);
                max = Math.max(val*dp[i-1][j][0], val*dp[i-1][j][1]);
                max = Math.max(max, val*dp[i][j-1][0]);
                max = Math.max(max, val*dp[i][j-1][1]);
            }
        }
        return new long[]{min, max};
    }
}
```
