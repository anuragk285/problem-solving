# Equal Sum Grid Partition I

```java
class Solution {
    public boolean canPartitionGrid(int[][] grid) {
        // finding totalSum of the matrix;
        long totalSum = 0;
        int ROWS = grid.length, COLS = grid[0].length;
        long[] rowsSum = new long[ROWS];
        long[] colsSum = new long[COLS];
        for(int i=0; i<ROWS; i++){
            for(int j=0; j<COLS; j++){
                rowsSum[i] += grid[i][j];
                colsSum[j] += grid[i][j];
                totalSum += grid[i][j];
            }
        }
        for(int i=1; i<ROWS; i++) rowsSum[i] += rowsSum[i-1];
        for(int j=1; j<COLS; j++) colsSum[j] += colsSum[j-1];
        // for horizontal cut
        for(int i=0; i<ROWS-1; i++){
            if(rowsSum[i] == (totalSum - rowsSum[i])) return true;
        }
        for(int j=0; j<COLS-1; j++){
            if(colsSum[j] == (totalSum - colsSum[j])) return true;
        }
        return false;
    }
}
```
