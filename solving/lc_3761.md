# Minimum Absolute Distance Between Mirror Pairs

```java
class Solution {
    public int minMirrorPairDistance(int[] nums) {
        HashMap<Integer, ArrayList<Integer>> indices = new HashMap<>();
        for(int i=0; i<nums.length; i++){
            indices.putIfAbsent(nums[i], new ArrayList<Integer>());
            indices.get(nums[i]).add(i);
        }
        int ans = Integer.MAX_VALUE;
        for(int i=0; i<nums.length; i++){
            int n = nums[i];
            int rev = reverse(n);
            //System.out.println("n = "+n+", rev(n) = "+rev);
            if(indices.containsKey(rev)){
                int iIL = -1; // indexInList
                int size = indices.get(rev).size();
                if(indices.get(rev).get(size-1) <= i) continue;
                if(indices.get(rev).get(0) > i){
                    ans = Math.min(ans, Math.abs(indices.get(rev).get(0) - i));
                }else{
                    int low =0, high = size-1;
                    while(low <= high){
                        int mid = low + (high - low)/2;
                        if(indices.get(rev).get(mid) > i){
                            iIL = mid;
                            high = mid-1;
                        }else{
                            low = mid+1;
                        }
                    }
                    if(iIL != -1) 
                        ans = Math.min(ans, indices.get(rev).get(iIL) - i);
                }
            }
        }
        return ans == Integer.MAX_VALUE ? -1 : ans;
    }
    private int reverse(int n){
        int rev = 0;
        while(n > 0){
            rev = (rev*10) + (n%10);
            n /= 10;
        }
        return rev;
    }
}
```
