# Minimum Distance Between Three Equal Elements I

```java
class Solution {
    public int minimumDistance(int[] nums) {
        HashMap<Integer, ArrayList<Integer>> map = new HashMap<>();
        for(int i=0; i<nums.length; i++){
            map.putIfAbsent(nums[i], new ArrayList<Integer>());
            map.get(nums[i]).add(i);
        }
        int ans = Integer.MAX_VALUE;
        for(int n : map.keySet()){
            if(map.get(n).size() < 3) continue;
            int x = 0;
            while(x+3 <= map.get(n).size()){
                int i = map.get(n).get(0+x), j = map.get(n).get(1+x), k = map.get(n).get(2+x);
                int d = Math.abs(i-j) + Math.abs(j-k) + Math.abs(k-i);
                ans = Math.min(ans, d);
                x++;
            }
        }
        return ans == Integer.MAX_VALUE ? -1 : ans;
    }
}
```
