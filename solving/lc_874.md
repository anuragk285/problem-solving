# Walking robot simulation

```java
class Solution {
        HashMap<Integer, ArrayList<Integer>> obstaclesOnX = new HashMap<>();
        HashMap<Integer, ArrayList<Integer>> obstaclesOnY = new HashMap<>();
        int x = 0, y = 0;
        char dir = 'N';
    public int robotSim(int[] commands, int[][] obstacles) {
        int ans = 0;
        for(int[] ob : obstacles){
            obstaclesOnX.putIfAbsent(ob[0], new ArrayList<>());
            obstaclesOnX.get(ob[0]).add(ob[1]);
            obstaclesOnY.putIfAbsent(ob[1], new ArrayList<>());
            obstaclesOnY.get(ob[1]).add(ob[0]);
        }
        // System.out.println(obstaclesOnX);
        // System.out.println(obstaclesOnY);
        for(int c : commands){
            if(1 <= c && c <= 9) moveForward(c);
            else if(c == -1){
                switch(dir){
                    case 'N' : {dir = 'R'; break;}
                    case 'R' : {dir = 'S'; break;}
                    case 'S' : {dir = 'W'; break;}
                    case 'W' : {dir = 'N'; break;}
                }
            }
            else if(c == -2){
                 switch(dir){
                    case 'N' : {dir = 'W'; break;}
                    case 'R' : {dir = 'N'; break;}
                    case 'S' : {dir = 'R'; break;}
                    case 'W' : {dir = 'S'; break;}
                }
            }
            ans = Math.max(ans, (x*x + y*y));
            //System.out.println("x = "+x+", y = "+y+", dir = "+dir+", c = "+c);
        }
        return ans;
    }
    private void moveForward(int k){
        switch(dir){
            case 'N' : {
                int nextY = y + k;
                if(obstaclesOnX.containsKey(x)){
                    for(int obY : obstaclesOnX.get(x)){
                        if(y < obY && obY <= nextY) nextY = obY-1;
                    }
                }
                y = nextY;
                break;
            }
            case 'R' : {
                int nextX = x + k;
                if(obstaclesOnY.containsKey(y)){
                    for(int obX : obstaclesOnY.get(y)){
                        if(x < obX && obX <= nextX) nextX = obX-1;
                    }
                }
                x = nextX;
                break;
            }
            case 'S' : {
                int nextY = y - k;
                if(obstaclesOnX.containsKey(x)){
                    for(int obY : obstaclesOnX.get(x)){
                        if(y > obY && obY >= nextY) nextY = obY+1;
                    }
                }
                y = nextY;
                break;
            }
            case 'W' : {
                int nextX = x - k;
                if(obstaclesOnY.containsKey(y)){
                    for(int obX : obstaclesOnY.get(y)){
                        if(x > obX && obX >= nextX) nextX = obX+1;
                    }
                }
                x = nextX;
                break;
            }
        }
    }
}
```
