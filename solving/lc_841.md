# Keys and Rooms

```java
class Solution {
    List<List<Integer>> rooms;
    boolean[] visited;
    public boolean canVisitAllRooms(List<List<Integer>> rooms) {
        this.rooms = rooms;
        visited = new boolean[rooms.size()];
        bfs(0);
        int sum = 0;
        for(boolean b : visited) if(b) sum++;
        return sum == rooms.size();
    }
    private void bfs(int r){
        if(visited[r]) return;
        visited[r] = true;
        for(int nextR : rooms.get(r)){
            if(!visited[nextR]) bfs(nextR);
        }
    }
}
```
