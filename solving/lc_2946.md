# Matrix Similarity after cyclic shifts

```java
class Solution {
    int[][] mat;
    int ROWS, COLS;
    public boolean areSimilar(int[][] mat, int k) {
        this.mat = mat;
        ROWS = mat.length;
        COLS = mat[0].length;
        int[][] org = new int[ROWS][COLS];
        k = k%COLS;
        for(int i=0; i<ROWS; i++) for(int j=0; j<COLS; j++) org[i][j] = mat[i][j];
        for(int r=0; r<ROWS; r++){
            for(int c=0; c<COLS; c++){
                int colIndex = r%2==0 ? (c + k + COLS)%COLS : (c - k + COLS)%COLS;
                mat[r][c] = org[r][colIndex];
                if(org[r][c] != mat[r][c]) return false;
            }
        }
        return true;
    }

}
```
