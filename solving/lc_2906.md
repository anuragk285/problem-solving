# Construct Product Matrix

```java
class Solution {
    public int[][] constructProductMatrix(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        long[][] prefix = new long[m][n];
        long[][] suffix = new long[m][n];
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(j == 0){
                    if(i == 0) prefix[i][j] = 1;
                    else prefix[i][j] = (prefix[i-1][n-1] * grid[i-1][n-1])%12345;
                }
                else prefix[i][j] = (prefix[i][j-1] * grid[i][j-1])%12345;
            }
        }
        for(int i=m-1; i>=0; i--){
            for(int j=n-1; j>=0; j--){
                if(j == n-1){
                    if(i == m-1) suffix[i][j] = 1;
                    else suffix[i][j] = (suffix[i+1][0] * grid[i+1][0])%12345;
                }
                else suffix[i][j] = (suffix[i][j+1] * grid[i][j+1])%12345;
            }
        }
        int[][] p = new int[m][n];
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                p[i][j] =(int)((prefix[i][j] * suffix[i][j]) % 12345);
            }
        }
        return p;
    }
}
```
