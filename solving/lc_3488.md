# Closest equal element queries

```java
class Solution {
    HashMap<Integer, ArrayList<Integer>> indices = new HashMap<>();
    public List<Integer> solveQueries(int[] nums, int[] queries) {
        int length = nums.length;
        for(int i=0; i<nums.length; i++){
            int n = nums[i];
            indices.putIfAbsent(n, new ArrayList<Integer>());
            indices.get(n).add(i);
        }
        List<Integer> ans = new ArrayList<>();
        for(int queryIndex : queries){
            int q = nums[queryIndex];
            //System.out.println("qIndex = "+queryIndex+", q = "+q);
            if(indices.get(q).size() == 1) ans.add(-1);
            else{
               int indexInList = binarySearch(q, queryIndex);
               int minDis = Integer.MAX_VALUE, size = indices.get(q).size();
               if(indexInList == 0){
                    minDis = indices.get(q).get(1) - indices.get(q).get(0);
                    minDis = Math.min(minDis, length - (indices.get(q).get(size-1) - indices.get(q).get(0)));
               }else if(0 < indexInList && indexInList < size-1){
                    int iIL = indexInList;
                    minDis = indices.get(q).get(iIL+1) - indices.get(q).get(iIL);
                    minDis = Math.min(minDis, (indices.get(q).get(iIL) - indices.get(q).get(iIL-1)));
               }else if(indexInList == size-1){
                    int iIL = indexInList;
                    minDis = indices.get(q).get(iIL) - indices.get(q).get(iIL-1);
                    minDis = Math.min(minDis, length - (indices.get(q).get(size-1) - indices.get(q).get(0)));
               }
               ans.add(minDis);
            }
        }
        return ans;
    }
    private int binarySearch(int q, int queryIndex){
        int low = 0, high = indices.get(q).size()-1;
        while(low <= high){
            int mid = low + (high - low)/2;
            if(indices.get(q).get(mid) == queryIndex) return mid;
            else if(indices.get(q).get(mid) < queryIndex) low = mid+1;
            else high = mid-1;
        }
        return -1;
    }
}
```
