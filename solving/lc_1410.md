# HTML Entity Parser

```java
class Solution {
    public String entityParser(String text) {
        String ans = "";
        int lastAmpIndex = -1;
        for(int i=0; i<text.length(); i++){
            ans += text.charAt(i);
            if(text.charAt(i) == '&') lastAmpIndex = ans.length()-1;
            else if(text.charAt(i) == ';'){
                if(lastAmpIndex == -1) continue;
                String s = ans.substring(lastAmpIndex, ans.length());
                String parsedChar = parseCode(s);
                if(!parsedChar.equals("")){
                    ans = ans.substring(0, lastAmpIndex) + parsedChar;
                    lastAmpIndex = -1;
                }
            }
        }
        return ans;
    }
    private String parseCode(String s){
        switch(s){
            case "&quot;" : return "" + '"';
            case "&apos;" : return "'";
            case "&amp;"  : return "" + '&';
            case "&gt;"   : return "" + '>';
            case "&lt;"   : return "" + '<';
            case "&frasl;": return "" + '/';
        }
        return "";
    }
}
```
